\begin{frame}{Model Predictive Control (MPC)}
    \begin{itemize}
        \item Problem $(\mathcal{S,A,P,R,\textcolor{mygreen}{C}})$\\
        \vspace{34mm}
        \visible<1->{
            \begin{textblock*}{3cm}(1.4cm,2cm)
                \begin{align*}
                    \textcolor{lightblue}{\max_{\V{a}}} \quad &\textcolor{lejla}{\mathbb{E}\left[\sum\limits^{N}_{k=0}\mathcal{R}(\V{s}_{k}, \V{a}_{k})\right]} \\
                    \text{s.t.} \quad & \V{s}_{k+1} = \textcolor{orange}{\V{f}}(\V{s}_{k}, \V{a}_{k}) \quad \forall k,\\
                    & -\V{b}<\V{a}_{k}<\V{b} \quad \forall k,\\
                    & \M{M}_{s_k}\V{s}_{k} \leq B_{s_k} \quad \forall k.
                \end{align*}
            \end{textblock*}
            \begin{textblock*}{3cm}(2.45cm,4.8cm)
                \begin{tikzpicture}
                    \draw[ultra thick, mygreen] (1.5, 0.8) -- (4.6, 0.8);
                    \draw[ultra thick, mygreen] (1.5, 1.45) -- (4.6, 1.45);
                \end{tikzpicture}
            \end{textblock*}
        }
        \pause
        \item Pros \& Cons\\
            \pause
            \textcolor{myred}{$\boldsymbol{+}$} safety (constraints)\\
            \pause
            \pause
            \textcolor{myblue}{$\boldsymbol{-}$} limited horizon $N$\\
            \pause
            \pause
            \textcolor{myblue}{$\boldsymbol{-}$} sub-optimal (quadratic problem)\\
            $\quad\; \rightarrow$ $\V{f}\sim\mathcal{P}$\pause, $N\ll\infty$\pause, $deg(\mathcal{C}(\cdot))=1$, $deg(\mathcal{R}(\cdot))=2$
    \end{itemize}

\begin{frame}{Reinforcement Learning (RL)}
    \begin{itemize}
    \item Problem $(\mathcal{S,A,P,R}, \gamma)$\\
    \vspace{35mm}
    \visible<1->{
        \begin{textblock*}{3cm}(1.4cm,2cm)
            \begin{align*}
                G=\sum_{k=0}^\infty\gamma^k\mathbb{E}_{(\V{s}_k,\V{a}_k)\sim \rho}\mathcal{R}(\V{s}_k,\V{a}_k)\\
            \end{align*}
        \end{textblock*}
    }
        \begin{textblock*}{3cm}(1.4cm,3.4cm)
        \visible<2->{
            \begin{align*}
                \pi^\star &= \argmax_\pi G(\pi), \; \text{where}\\
                &G(\pi)=\mathbb{E}_{\V{\tau}\sim \rho_\pi}\left[\sum_{k=0}^\infty\gamma^k\mathcal{R}(\V{s}_k,\V{a}_k)\right]\\
            \end{align*}
        }
        \end{textblock*}
    \pause
    \pause
    \item Pros \& Cons\\
        \pause
        \textcolor{myred}{$\boldsymbol{+}$} learn from data\\
        \pause
        \textcolor{myred}{$\boldsymbol{+}$} infinite horizon\\
        \pause
        \textcolor{myblue}{$\boldsymbol{-}$} safety\\
    \end{itemize}

\begin{frame}{Trust Region Projection Layers (TRPL)}
    \begin{itemize}
        \item Trust regions\\
            - stabilize learning\\
            \hspace{5mm}$\M{\theta}^\star= \argmax_{\M{\theta}}G(\pi_{\M{\theta}},\pi_{\M{\theta}_{old}})$\\
            \hspace{10mm}$\text{s.t.}\quad D(\pi_{\M{\theta}_{old}}, \pi_{\M{\theta}}) < \delta,\quad \forall \V{s}\in\mathcal{S}$
        \pause
        \item Distance metric\\
            - Kullback-Leibler\\
            \hspace{5mm}$D(\pi_{\M{\theta}}\vert\vert \pi_{\M{\theta}_{old}}) = \sum\limits_{\V{s},\V{a}}\pi_{\M{\theta}}(\V{a}\vert \V{s})\log\dfrac{\pi_{\M{\theta}}(\V{a}\vert\V{s})}{\pi_{\M{\theta}_{old}}(\V{a}\vert\V{s})}<\delta$
        \pause
        \item Derive mean and variance projection\\
            \hspace{5mm}$\argmin\limits_{\tilde{\mu}} \T{(\mu-\tilde{\mu})}\Sigma_{old}^{-1}(\mu-\tilde{\mu}),\quad \text{s.t.}\quad \T{(\mu_{old}-\tilde{\mu})}\Sigma_{old}^{-1}(\mu_{old}-\tilde{\mu})\leq\delta_\mu$\\
            \vspace{2mm}
            \hspace{40mm}$\rightarrow \tilde\mu=\dots, \quad\tilde\Sigma=\dots $
    \end{itemize}
