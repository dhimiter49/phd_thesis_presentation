                    \textcolor{lightblue}{\max_{\V{a}}} \quad &\textcolor{lejla}{\mathbb{E}\left[\sum^{N}_{k=0}\mathcal{R}(\V{s}_{k}, \V{a}_{k})\right]} \\
                    \text{s.t.} \quad & \V{s}_{k+1} = \textcolor{orange}{\V{f}}(\V{s}_{k}, \V{a}_{k}), \quad -\V{b}<\V{a}_{k}<\V{b} \quad \forall k.

                \min\limits_{\dot{P}} \quad & go\_to\_goal(\dot{P})\\

\begin{frame}[label={intro}]
\frametitle{Model Predictive Control (MPC)}
    \begin{itemize}
        \item Problem $(\mathcal{S,A,P,R,\textcolor{mygreen}{C}})$\\
        \vspace{34mm}
        \visible<1->{
            \begin{textblock*}{3cm}(1.4cm,2cm)
                \begin{align*}
                    \textcolor{lightblue}{\max_{\V{a}}} \quad &\textcolor{lejla}{\mathbb{E}\left[\sum\limits^{N}_{k=0}\mathcal{R}(\V{s}_{k}, \V{a}_{k})\right]} \\
                    \text{s.t.} \quad & \V{s}_{k+1} = \textcolor{orange}{\V{f}}(\V{s}_{k}, \V{a}_{k}) \quad \forall k,\\
                    & -\V{b}<\V{a}_{k}<\V{b} \quad \forall k,\\
                    & \M{M}_{s_k}\V{s}_{k} \leq B_{s_k} \quad \forall k.
                \end{align*}
            \end{textblock*}
            \begin{textblock*}{3cm}(2.45cm,4.8cm)

                \end{tikzpicture}
            \end{textblock*}
        }
        \pause
        \item Pros \& Cons\\
            \pause
            \textcolor{myred}{$\boldsymbol{+}$} safety (constraints)\\
            \pause
            \pause
            \textcolor{myblue}{$\boldsymbol{-}$} limited horizon $N$\\
            \pause
            \pause
            \textcolor{myblue}{$\boldsymbol{-}$} approximation\\
            $\quad\; \rightarrow$ $\V{f}\sim\mathcal{P}$\pause, $N\ll\infty$\pause, $deg(\mathcal{C}(\cdot))=1$, $deg(\mathcal{R}(\cdot))=2$

\begin{frame}[label={intro}]
\frametitle{Reinforcement Learning (RL)}
    \begin{itemize}
    \item Problem $(\mathcal{S,A,P,R}, \gamma)$\\
    \vspace{35mm}
    \visible<1->{
        \begin{textblock*}{3cm}(1.4cm,2cm)
            \begin{align*}
                G=\sum_{k=0}^\infty\gamma^k\mathbb{E}_{(\V{s}_k,\V{a}_k)\sim \rho}\mathcal{R}(\V{s}_k,\V{a}_k)\\
            \end{align*}
        \end{textblock*}
    }
        \begin{textblock*}{3cm}(1.4cm,3.4cm)
        \visible<2->{
            \begin{align*}
                \pi^\star &= \argmax_\pi G(\pi), \; \text{where}\\
                &G(\pi)=\mathbb{E}_{\V{\tau}\sim \rho_\pi}\left[\sum_{k=0}^\infty\gamma^k\mathcal{R}(\V{s}_k,\V{a}_k)\right]\\
            \end{align*}
        }
        \end{textblock*}
    \pause
    \pause
    \item Pros \& Cons\\
        \pause
        \textcolor{myred}{$\boldsymbol{+}$} learn from data\\
        \pause
        \textcolor{myred}{$\boldsymbol{+}$} infinite horizon\\
        \pause
        \textcolor{myblue}{$\boldsymbol{-}$} safety\\
    \end{itemize}

    \nabla_\theta G(\theta)&=\nabla_\theta\mathbb{E}_{\tau\sim p_\theta}\sum_t \gamma^t r(s_t,a_t)\\
    &= \nabla_\theta\int p_\theta(\tau)R(\tau)d\tau= \int \nabla_\theta p_\theta(\tau)R(\tau)d\tau=\int p_\theta(\tau) \nabla_\theta \log p_\theta(\tau) R(\tau)d\tau\\
    &= \mathbb{E}_{\tau\sim p_\theta(\tau)} \left[\left(\sum_t\nabla_\theta\pi_\theta(a_{t}\vert s_{t})\right)\left(\sum_{k=0}\gamma^{k}r(s_{k},a_{k})\right)\right]\\
