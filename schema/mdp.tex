\begin{tikzpicture}[every node/.append style={ultra thick},
    state/.style = {circle, draw=black, fill=white, minimum size=0.9cm},
    L/.style = {draw, black, line width=2, -{stealth[scale=2]}}]

    % fixed reference frame: pin the bounding box so the picture size
    % (and the \resizebox scale) does not change when arrows/values
    % appear per overlay or more content is added later
    \useasboundingbox (-5.5,-2.4) rectangle (5.5,3.5);

    % three states in a wide, squashed triangle
    \node<8->[state] (s1) at (-3, 0) {$\V{s}_1$};
    \node<8->[state] (s2) at (3, 0) {$\V{s}_2$};
    \node<8->[state] (s3) at (0, 1.4) {$\V{s}_3$};

    % bottom connections: two arcs of different depth
    \draw<9->[L, draw=myred] (s1) to[bend right=40] node[below, pos=0.5] {$\visible<11->{0.4\vert}{\color{mygreen}\visible<10->{+1}}$} (s2);
    \draw<9->[L, draw=myred] (s2) to[bend left=15] node[below, pos=0.5] {$\visible<11->{0.8\vert}{\color{mygreen}\visible<10->{-1}}$} (s1);

    % left connections: one arc inside, one outside
    \draw<9->[L, draw=myblue] (s1) to[bend right=22] node[above, pos=0.35, sloped] {$\visible<11->{0.1\vert}{\color{mygreen}\visible<10->{+2}}$} (s3);
    \draw<9->[L, draw=myred] (s3) to[bend right=22] node[above left, pos=0.5] {$\visible<11->{0.19\vert}{\color{mygreen}\visible<10->{0}}$} (s1);

    % right connections: one arc inside, one outside
    \draw<9->[L, draw=myblue] (s2) to[bend left=22] node[above, pos=0.35, sloped] {$\visible<11->{0.1\vert}{\color{mygreen}\visible<10->{+3}}$} (s3);
    \draw<9->[L, draw=myred] (s3) to[bend left=22] node[above right, pos=0.5] {$\visible<11->{0.8\vert}{\color{mygreen}\visible<10->{-2}}$} (s2);

    % self-loops
    \draw<9->[L, draw=myblue] (s1) edge[loop left] node[left] {$\visible<11->{0.5\vert}{\color{mygreen}\visible<10->{0}}$} (s1);
    \draw<9->[L, draw=myred] (s2) edge[loop right] node[right] {$\visible<11->{0.1\vert}{\color{mygreen}\visible<10->{0}}$} (s2);
    \draw<9->[L, draw=myblue] (s3) edge[loop above] node[above] {$\visible<11->{0.01\vert}{\color{mygreen}\visible<10->{+5}}$} (s3);
\end{tikzpicture}
